    private void cycle(long runGeneration) {
        if (!active
            || runGeneration != generation
            || !inIteration.compareAndSet(false, true)) {
            return;
        }

        try {
            if (!initialized) {
                initialize();
            }

            if (!canAct() || terminal()) {
                return;
            }

            Snapshot snapshot = snapshot();

            if (snapshot == null) {
                handleUnavailableClient();
                return;
            }

            if (!bindAccount(snapshot)) {
                return;
            }

            if (offlineSince != 0) {
                offlineSince = 0;

                if (config.enableCrashRecovery()) {
                    abort(
                        "Connection interrupted during run",
                        snapshot.location
                    );
                } else {
                    finish(
                        State.FAILED,
                        "Connection interrupted; crash recovery disabled"
                    );
                    return;
                }
            }

            if (!super.run()) {
                clearRoute();
                lastMovement = now();
                return;
            }

            /*
             * Recovery is checked at every worker-loop entry with a valid
             * snapshot, before any state-specific input.
             */
            if (config.enableCrashRecovery()
                && !expectedZone(snapshot.location)) {
                abort(
                    "Outside expected state zone: " + snapshot.location,
                    snapshot.location
                );
            }

            if (terminal() || !canAct()) {
                return;
            }

            if (!snapshot.location.equals(lastLocation)) {
                lastLocation = snapshot.location;
                lastMovement = now();
                stuckRetries = 0;
            }

            if (now() - stateSince > stateBudget()) {
                if (state == State.HOMING_TO_BANK
                    || state == State.DEPOSITING_LOOT) {
                    finish(
                        State.FAILED,
                        "Recovery/banking deadline exceeded: " + failure
                    );
                } else {
                    abort(
                        "State deadline exceeded: " + state,
                        snapshot.location
                    );
                }
                return;
            }

            if (snapshot.hp <= 0) {
                abort(
                    "Player died; onboarding cannot be marked successful",
                    snapshot.location
                );
                return;
            }

            if (handleHealth(snapshot)) {
                return;
            }

            if (handleDialogue(snapshot)) {
                return;
            }

            switch (state) {
                case VALIDATING_ANCHOR:
                    validate(snapshot);
                    break;

                case WALKING_TO_STRONGHOLD:
                    if (walk(ENTRANCE_STAND, 2, snapshot)) {
                        transition(
                            State.INTERACTING_ENTRANCE,
                            "At Barbarian Village entrance"
                        );
                    }
                    break;

                case INTERACTING_ENTRANCE:
                    if (floor(snapshot.location) == 0) {
                        transition(
                            State.TRAVERSING_FLOOR_1,
                            "Observed first-floor arrival"
                        );
                    } else if (walk(ENTRANCE_STAND, 2, snapshot)) {
                        interactTransition(
                            ObjectID.SOS_DUNG_ENT_OPEN,
                            "Climb-down"
                        );
                    }
                    break;

                case TRAVERSING_FLOOR_1:
                    traverse(0, snapshot);
                    break;

                case TRAVERSING_FLOOR_2:
                    traverse(1, snapshot);
                    break;

                case TRAVERSING_FLOOR_3:
                    traverse(2, snapshot);
                    break;

                case TRAVERSING_FLOOR_4:
                    if (walk(REWARD_STANDS[3], 1, snapshot)) {
                        transition(
                            State.CLAIMING_REWARDS,
                            "At the Cradle of Life"
                        );
                    }
                    break;

                case CLAIMING_REWARDS:
                    claimBoots(snapshot);
                    break;

                case HOMING_TO_BANK:
                    if (walk(ANCHOR, 4, snapshot)) {
                        transition(
                            State.DEPOSITING_LOOT,
                            "Observed GE anchor arrival"
                        );
                    }
                    break;

                case DEPOSITING_LOOT:
                    bank(snapshot);
                    break;

                case COMPLETED:
                case FAILED:
                    break;

                default:
                    throw new IllegalStateException(
                        "Unhandled state " + state
                    );
            }
        } catch (Exception exception) {
            if (canAct()) {
                Microbot.log(
                    "[Stronghold] "
                        + exception.getClass().getSimpleName()
                        + ": "
                        + exception.getMessage()
                );

                finish(
                    State.FAILED,
                    "Unexpected exception; no further input: "
                        + exception.getMessage()
                );
            }
        } finally {
            inIteration.set(false);
        }
    }

    private void handleUnavailableClient() {
        if (offlineSince == 0) {
            offlineSince = now();
            clearRoute();
        } else if (now() - offlineSince >= OFFLINE_TIMEOUT_MS) {
            finish(

                "Logged out or client unavailable for 90 seconds"
            );
        }

    private boolean bindAccount(Snapshot snapshot) {
        if (accountName != null) {
            if (!accountName.equals(snapshot.name)) {
                finish(
                    State.FAILED,
                    "Account changed; refusing input on a different account"
                );
                return false;
            }

            return true;
        }

        accountName = snapshot.name;
        accountKey = "run_" + Base64.getUrlEncoder()
            .withoutPadding()
            .encodeToString(
                accountName.getBytes(StandardCharsets.UTF_8)
            );

        String previous = configManager.getConfiguration(
            StrongholdConfig.GROUP,
            accountKey
        );

        if (previous != null && previous.startsWith("COMPLETED|")) {
            if (snapshot.location.distanceTo(ANCHOR) > 6
                || snapshot.usedSlots != 0
                || snapshot.equipmentSlots != 0) {
                finish(
                    State.FAILED,
                    "Stage previously completed, but account is no longer "
                        + "empty at anchor"
                );
                return false;
            }

            Arrays.fill(receipts, true);
            bootsReceived = true;

            finish(
                State.COMPLETED,
                "Previously verified completion; account remains empty at anchor"
            );
            return false;
        }

        if (previous != null
            && (previous.startsWith("RUNNING|")
                || previous.startsWith("INTERRUPTED|"))) {
            if (!config.enableCrashRecovery()) {
                finish(
                    State.FAILED,
                    "Interrupted journal; crash recovery disabled"
                );
                return false;
            }

            abort(
                "Interrupted run detected; returning remaining items to bank",
                snapshot.location
            );
        }

        return true;
    }

    private boolean handleHealth(Snapshot snapshot) {
        if (floor(snapshot.location) < 0
            || snapshot.hp * 100 > snapshot.maxHp * 60) {
            return false;
        }

        boolean ate = Rs2Player.eatAt(60);
        actionDelay();

        if (ate) {
            return true;
        }

        if (state != State.HOMING_TO_BANK) {
            abort(
                "Low health with no usable food",
                snapshot.location
            );
            return true;
        }

        return false;
    }

    private void validate(Snapshot snapshot) {
        if (snapshot.location.distanceTo(ANCHOR) > 15) {
            abort(
                "Account did not start at the GE anchor",
                snapshot.location
            );
            return;
        }

        for (boolean claimed : snapshot.emotes) {
            if (claimed) {
                finish(
                    State.FAILED,
                    "One-time reward already claimed; use a fresh "
                        + "onboarding account"
                );
                return;
            }
        }

        if (snapshot.boots != 0
            || snapshot.usedSlots > 26
            || Rs2Inventory.getInventoryFood().size() < 4) {
            finish(
                State.FAILED,
                "Need four edible food slots, two free inventory slots "
                    + "and no Fancy boots"
            );
            return;
        }

        if (Rs2Bank.isOpen()) {
            Rs2Bank.closeBank();
            actionDelay();
            return;
        }

        initialCoins = snapshot.coins;

        transition(
            State.WALKING_TO_STRONGHOLD,
            "Validated fresh rewards, supplies and anchor"
        );
    }

    private void traverse(int level, Snapshot snapshot) {
        if (floor(snapshot.location) == level + 1) {
            if (!receipts[level]) {
                abort(
                    "Left floor before its coin receipt was verified",
                    snapshot.location
                );
            } else {
                transition(
                    FLOOR_STATES[level + 1],
                    "Observed next-floor arrival"
                );
            }

            return;
        }

        if (!receipts[level]) {
            if (walk(REWARD_STANDS[level], 1, snapshot)) {
                claimCoins(level, snapshot);
            }

            return;
        }

        if (walk(DESCENT_STANDS[level], 1, snapshot)) {
            interactTransition(
                DESCENT_IDS[level],
                "Climb-down"
            );
        }
    }

    private void claimCoins(int level, Snapshot snapshot) {
        if (snapshot.emotes[level]) {
            /*
             * Allow inventory and varbit events to settle before rejecting
             * a partially observed reward update.
             */
            if (pendingReward == level
                && snapshot.coins - rewardCoinsBefore != REWARD_GP[level]
                && now() - lastRewardClick < 8000) {
                return;
            }

            if (pendingReward != level
                || snapshot.coins - rewardCoinsBefore != REWARD_GP[level]) {
                abort(
                    "Reward emote/coin receipt mismatch on floor "
                        + (level + 1),
                    snapshot.location
                );
                return;
            }

            receipts[level] = true;
            pendingReward = -1;
            rewardSince = 0;
            lastRewardClick = 0;

            save(
                "Verified floor "
                    + (level + 1)
                    + " receipt: "
                    + REWARD_GP[level]
                    + " GP"
            );

            clearRoute();
            return;
        }

        if (pendingReward != level) {
            pendingReward = level;
            rewardCoinsBefore = snapshot.coins;
            rewardSince = now();
            lastRewardClick = 0;

            save(
                "Beginning reward claim on floor " + (level + 1)
            );
        }

        if (now() - rewardSince > 60000) {
            abort(
                "Reward not granted; check two-factor authentication "
                    + "and account eligibility",
                snapshot.location
            );
        } else if (lastRewardClick == 0
            || now() - lastRewardClick >= 10000) {
            interactReward(REWARD_IDS[level]);
        }
    }

    private void claimBoots(Snapshot snapshot) {
        if (snapshot.emotes[3] && snapshot.boots > 0) {
            if (!allCoinReceipts()
                || (long) snapshot.coins != (long) initialCoins + 10000) {
                abort(
                    "Final coin total does not match verified "
                        + "10,000 GP receipts",
                    snapshot.location
                );
                return;
            }

            bootsReceived = true;
            homeOrigin = ENTRANCE_STAND;

            transition(
                State.HOMING_TO_BANK,
                "Verified Fancy boots, four emotes and 10,000 GP"
            );
            return;
        }

        if (rewardSince == 0) {
            rewardSince = now();
        }

        if (now() - rewardSince > 90000) {
            abort(
                "Boot reward timed out; check two-factor authentication",
                snapshot.location
            );
        } else if (lastRewardClick == 0
            || now() - lastRewardClick >= 10000) {
            interactReward(REWARD_IDS[3]);
        }
    }

    private void interactReward(int id) {
        if (!canAct()
            || Rs2Player.isMoving()
            || Rs2Player.isAnimating()) {
            return;
        }

        TileObject object = Rs2GameObject.findObjectById(id);

        if (object == null) {
            return;
        }

        String action = Rs2GameObject.hasAction(object, "Search")
            ? "Search"
            : "Open";

        clearRoute();

        if (canAct()) {
            Rs2GameObject.interact(object, action);
            lastRewardClick = now();
            actionDelay();
        }
    }

    private void interactTransition(int id, String action) {
        if (!canAct()
            || now() - lastObjectClick < 8000
            || Rs2Player.isMoving()
            || Rs2Player.isAnimating()) {
            return;
        }

        if (++objectAttempts > 5) {
            abort(
                "Transition object did not produce an observed arrival",
                lastLocation
            );
            return;
        }

        clearRoute();

        transitWarningUntil = now() + 30000;

        Rs2GameObject.interact(id, action);

        lastObjectClick = now();
        actionDelay();
    }

    private boolean handleDialogue(Snapshot snapshot) {
        if (!Rs2Dialogue.isInDialogue()) {
            dialogueSince = 0;
            rememberedAnswer = null;
            dismissDialogue = false;
            return false;
        }

        if (dialogueSince == 0) {
            dialogueSince = now();
        }

        if (now() - dialogueSince > DIALOGUE_TIMEOUT_MS) {
            finish(
                State.FAILED,
                "Dialogue did not close within 90 seconds"
            );
            return true;
        }

        String text = normalize(Rs2Dialogue.getDialogueText())
            + " "
            + normalize(Rs2Dialogue.getQuestion());

        String answer = answerFor(text);

        if (answer != null) {
            rememberedAnswer = answer;
        }

        List<String> options = dialogueOptions();
        String selected = null;

        if (!options.isEmpty()) {
            if (state == State.CLAIMING_REWARDS) {
                for (String option : options) {
                    /*
                     * "fancier boots" does not contain "fancy boots".
                     * Selection is by verified visible text, not index.
                     */
                    if (normalize(option).contains("fancy boots")) {
                        selected = option;
                        break;
                    }
                }
            }

            if (selected == null && rememberedAnswer != null) {
                for (String option : options) {
                    if (normalize(option).equals(
                        normalize(rememberedAnswer)
                    )) {
                        selected = option;
                        break;
                    }
                }
            }

            String optionText = normalize(
                String.join(" ", options)
            );

            boolean transitionWarning = now() <= transitWarningUntil
                && (text.contains("danger")
                    || text.contains("sure")
                    || text.contains("continue")
                    || optionText.contains("danger"));

            if (selected == null && transitionWarning) {
                for (String option : options) {
                    String normalized = normalize(option);

                    if (normalized.equals("yes")
                        || normalized.equals("yes.")
                        || normalized.startsWith("yes,")) {
                        selected = option;
                        break;
                    }
                }
            }

            if (selected == null && !dismissDialogue) {
                dismissDialogue = true;

                abort(
                    "Unrecognized dialogue options; refusing to guess: "
                        + options,
                    snapshot.location
                );
            }
        }

        Global.sleep(Rs2Random.between(1800, 4200));

        if (!canAct() || !Rs2Dialogue.isInDialogue()) {
            return true;
        }

        /*
         * Revalidate the account after the reading delay, before issuing input.
         */
        Snapshot current = snapshot();

        if (current == null) {
            return true;
        }

        if (!accountName.equals(current.name)) {
            finish(
                State.FAILED,
                "Account changed during dialogue reading delay"
            );
            return true;
        }

        if (dismissDialogue) {
            Rs2Keyboard.keyPress(KeyEvent.VK_ESCAPE);
        } else if (selected != null) {
            if (dialogueOptions().contains(selected)
                && Rs2Dialogue.isInDialogue()) {
                Rs2Dialogue.clickOption(selected, true);
                rememberedAnswer = null;
            }
        } else if (Rs2Dialogue.hasContinue()
            && Rs2Dialogue.isInDialogue()) {
            Rs2Dialogue.clickContinue();
        }

        actionDelay();
        return true;
    }

    private String answerFor(String text) {
        String result = null;

        for (Map.Entry<String, String> entry
            : securityAnswers.entrySet()) {
            if (!text.contains(entry.getKey())) {
                continue;
            }

            if (result != null && !result.equals(entry.getValue())) {
                throw new IllegalStateException(
                    "Ambiguous security question"
                );
            }

            result = entry.getValue();
        }

        return result;
    }

    private List<String> dialogueOptions() {
        return Microbot.getClientThread()
            .runOnClientThreadOptional(() -> {
                List<String> result = new ArrayList<>();

                for (Widget option : Rs2Dialogue.getDialogueOptions()) {
                    result.add(option.getText());
                }

                return result;
            })
            .orElse(Collections.emptyList());
    }

    /**
     * Caller-driven path following. The first upcoming door or transport is
     * identified before issuing a walk step, so the walker never owns security
     * dialogue selection.
     */
    private boolean walk(
        WorldPoint destination,
        int radius,
        Snapshot snapshot
    ) {
        if (!canAct()) {
            return false;
        }

        if (pendingBarrier != null) {
            boolean crossed;

            if (pendingBarrier.transport) {
                crossed = snapshot.location.distanceTo(
                    pendingBarrier.to
                ) <= 3;
            } else {
                crossed = Rs2DoorGeometry.playerBeyondWallFace(
                    pendingBarrier.orientation,
                    pendingBarrier.location,
                    pendingBarrier.before,
                    snapshot.location
                );
            }

            if (crossed) {
                clearRoute();
            } else if (now() - barrierSince >= STUCK_MS) {
                if (++barrierAttempts > 4) {
                    abort(
                        "Door/transport failed to move player across "
                            + "its boundary",
                        snapshot.location
                    );
                    return false;
                }

                clickBarrier(pendingBarrier);
                return false;
            } else {
                return false;
            }
        }

        if (snapshot.location.distanceTo(destination) <= radius
            && Rs2GameObject.canReach(destination)) {
            clearRoute();
            return true;
        }

        if (!destination.equals(goal) || route.isEmpty()) {
            goal = destination;
            stepGoal = null;

            route = Rs2Walker.getWalkPath(
                snapshot.location,
                destination
            );

            if (route == null || route.isEmpty()) {
                route = Collections.emptyList();
                checkStuck(snapshot);
                return false;
            }
        }

        if (checkStuck(snapshot) || !canAct()) {
            return false;
        }

        int index = nearestIndex(
            route,
            snapshot.location
        );

        if (index < 0
            || snapshot.location.distanceTo(route.get(index)) > 3) {
            clearRoute();
            return false;
        }

        State routeState = state;
        Barrier barrier = findBarrier(index, snapshot);

        /*
         * findBarrier can abort and clear the route if it encounters an
         * unsupported transport. Never continue using that invalidated route.
         */
        if (state != routeState || goal == null || !canAct()) {
            return false;
        }

        int end = Math.min(
            route.size() - 1,
            index + 8
        );

        if (barrier != null) {
            boolean nearBarrier =
                snapshot.location.distanceTo(barrier.location) <= 2
                    && snapshot.location.distanceTo(barrier.from) <= 2;

            if (nearBarrier) {
                if (Rs2Player.isMoving() || Rs2Player.isAnimating()) {
                    return false;
                }

                pendingBarrier = barrier;
                pendingBarrier.before = snapshot.location;
                barrierAttempts = 0;

                clickBarrier(barrier);
                return false;
            }

            /*
             * The walk target must remain on the near side of the barrier.
             */
            end = Math.min(end, barrier.index);
        }

        WorldPoint base = route.get(end);

        if (base.equals(snapshot.location) && barrier != null) {
            abort(
                "Unable to approach the next route barrier safely",
                snapshot.location
            );
            return false;
        }

        if (stepGoal == null
            || snapshot.location.distanceTo(stepGoal) <= 1
            || stepGoal.distanceTo(base) > 2) {
            stepGoal = jitter(base);
        }

        if (canAct()) {
            Rs2Walker.walkStep(stepGoal, 0);
            actionDelay();
        }

        return false;
    }

    private Barrier findBarrier(
        int index,
        Snapshot snapshot
    ) {
        List<TileObject> doors = Microbot.getClientThread()
            .runOnClientThreadOptional(() -> {
                List<TileObject> found = new ArrayList<>(
                    Rs2GameObject.getAll(
                        object -> DOOR_IDS.contains(object.getId())
                    )
                );

                found.sort(
                    Comparator.comparingInt(TileObject::getId)
                        .thenComparingInt(
                            object -> object.getWorldLocation().getX()
                        )
                        .thenComparingInt(
                            object -> object.getWorldLocation().getY()
                        )
                );

                return found;
            })
            .orElse(Collections.emptyList());

        int limit = Math.min(
            route.size() - 1,
            index + 16
        );

        for (int i = index; i < limit; i++) {
            WorldPoint from = route.get(i);
            WorldPoint to = route.get(i + 1);

            if (from.distanceTo(to) > 1) {
                List<Transport> links =
                    Rs2Walker.getTransportsForPath(
                        Arrays.asList(from, to),
                        0
                    );

                for (Transport link : links) {
                    if (!from.equals(link.getOrigin())
                        || !to.equals(link.getDestination())
                        || !LINK_IDS.contains(link.getObjectId())) {
                        continue;
                    }

                    TileObject object =
                        Rs2GameObject.findObjectById(
                            link.getObjectId()
                        );

                    WorldPoint location = object == null
                        ? from
                        : object.getWorldLocation();

                    return new Barrier(
                        i,
                        from,
                        to,
                        location,
                        link.getObjectId(),
                        link.getAction(),
                        true,
                        0
                    );
                }

                abort(
                    "Unsupported route transport; refusing automatic "
                        + "teleport or unknown interaction",
                    snapshot.location
                );
                return null;
            }

            for (TileObject door : doors) {
                if (door instanceof WallObject
                    && Rs2DoorGeometry.isDoorOnSegment(
                        door,
                        from,
                        to
                    )) {
                    return new Barrier(
                        i,
                        from,
                        to,
                        door.getWorldLocation(),
                        door.getId(),
                        "Open",
                        false,
                        doorOrientation(
                            (WallObject) door,
                            from,
                            to
                        )
                    );
                }
            }
        }

        return null;
    }

    private static int doorOrientation(
        WallObject door,
        WorldPoint from,
        WorldPoint to
    ) {
        boolean crossesA =
            Rs2DoorGeometry.playerBeyondWallFace(
                door.getOrientationA(),
                door.getWorldLocation(),
                from,
                to
            );

        return crossesA
            ? door.getOrientationA()
            : door.getOrientationB();
    }

    private void clickBarrier(Barrier barrier) {
        if (!canAct()) {
            return;
        }

        Rs2Walker.setTarget(
            null,
            "stage0:barrier"
        );

        if (barrier.transport) {
            transitWarningUntil = now() + 30000;
        }

        TileObject object = Microbot.getClientThread()
            .runOnClientThreadOptional(() ->
                Rs2GameObject.getAll(
                    candidate -> candidate.getId() == barrier.id
                        && candidate.getWorldLocation().distanceTo(
                            barrier.location
                        ) <= 1
                )
                    .stream()
                    .findFirst()
                    .orElse(null)
            )
            .orElse(null);

        if (object != null && canAct()) {
            Rs2GameObject.interact(
                object,
                barrier.action
            );
        }

        barrierSince = now();
        lastMovement = now();

        actionDelay();
    }

    private boolean checkStuck(Snapshot snapshot) {
        if (now() - lastMovement < STUCK_MS) {
            return false;
        }

        lastMovement = now();

        if (++stuckRetries > 5) {
            if (state == State.HOMING_TO_BANK) {
                finish(
                    State.FAILED,
                    "Unable to reach bank after five stationary "
                        + "recovery attempts"
                );
            } else {
                abort(
                    "Walking made no progress after five recovery attempts",
                    snapshot.location
                );
            }
        } else {
            Microbot.log(
                "[Stronghold] Stationary for 8+ seconds; "
                    + "clearing and reissuing route ("
                    + stuckRetries
                    + "/5)"
            );

            clearRoute();
        }

        return true;
    }

    private WorldPoint jitter(WorldPoint base) {
        if (!config.enableJitter()) {
            return base;
        }

        /*
         * between() excludes its upper bound in the pinned revision.
         * [-1, 2) therefore supplies the requested -1, 0, +1 offsets.
         */
        WorldPoint candidate = new WorldPoint(
            base.getX() + Rs2Random.between(-1, 2),
            base.getY() + Rs2Random.between(-1, 2),
            base.getPlane()
        );

        boolean safe = Microbot.getClientThread()
            .runOnClientThreadOptional(() ->
                Rs2Walker.isWalkableInCollisionMap(candidate)
                    && Rs2GameObject.canReach(candidate)
            )
            .orElse(false);

        /*
         * Safety takes precedence over jitter when an offset is blocked,
         * unloaded, or across a closed door.
         */
        return safe ? candidate : base;
    }

    private void bank(Snapshot snapshot) {
        if (snapshot.location.distanceTo(ANCHOR) > 6) {
            abort(
                "Player left the bank before deposit verification",
                snapshot.location
            );
            return;
        }

        if (bankVerified) {
            if (Rs2Bank.isOpen()) {
                Rs2Bank.closeBank();
                actionDelay();
                return;
            }

            if (snapshot.usedSlots != 0
                || snapshot.equipmentSlots != 0) {
                finish(
                    State.FAILED,
                    "Items changed after bank verification"
                );
                return;
            }

            boolean success = failure == null
                && allCoinReceipts()
                && bootsReceived;

            finish(
                success ? State.COMPLETED : State.FAILED,
                success
                    ? "All reward receipts and bank deltas verified; "
                        + "inventory/equipment empty"
                    : "Recovered to GE and deposited remaining items: "
                        + failure
            );
            return;
        }

        if (!Rs2Bank.isOpen()) {
            Rs2Bank.openBank();
            actionDelay();
            return;
        }

        if (!bankBaselineSet) {
            bankCoinsBefore = Rs2Bank.count(ItemID.COINS);
            bankBootsBefore = Rs2Bank.count(ItemID.SOS_BOOTS);

            depositingCoins = snapshot.coins;
            depositingBoots = snapshot.boots;

            bankBaselineSet = true;

            if (failure == null
                && (!allCoinReceipts()
                    || !bootsReceived
                    || (long) snapshot.coins
                        != (long) initialCoins + 10000
                    || snapshot.boots < 1)) {
                failure = "Reward inventory changed before deposit";
            }
        }

        if (snapshot.usedSlots > 0) {
            Rs2Bank.depositAll();
            actionDelay();
            return;
        }

        if (snapshot.equipmentSlots > 0) {
            Rs2Bank.depositEquipment();
            actionDelay();
            return;
        }

        boolean coinsDeposited =
            (long) Rs2Bank.count(ItemID.COINS)
                >= (long) bankCoinsBefore + depositingCoins;

        boolean bootsDeposited =
            (long) Rs2Bank.count(ItemID.SOS_BOOTS)
                >= (long) bankBootsBefore + depositingBoots;

        if (!coinsDeposited || !bootsDeposited) {
            return;
        }

        bankVerified = true;

        save(
            "Verified inventory/equipment emptied and "
                + "coin/boot bank increases"
        );
    }

    private boolean expectedZone(WorldPoint location) {
        switch (state) {
            case VALIDATING_ANCHOR:
                return location.distanceTo(ANCHOR) <= 15;

            case WALKING_TO_STRONGHOLD:
                return surfaceZone(
                    location,
                    ENTRANCE_STAND
                );

            case INTERACTING_ENTRANCE:
                return location.distanceTo(ENTRANCE_STAND) <= 15
                    || nearFloor(location, 0);

            case TRAVERSING_FLOOR_1:
            case TRAVERSING_FLOOR_2:
            case TRAVERSING_FLOOR_3:
                int level = state.ordinal()
                    - State.TRAVERSING_FLOOR_1.ordinal();

                return nearFloor(location, level)
                    || (receipts[level]
                        && nearFloor(location, level + 1));

            case TRAVERSING_FLOOR_4:
            case CLAIMING_REWARDS:
                return nearFloor(location, 3);

            case HOMING_TO_BANK:
                for (int i = 0; i < FLOORS.length; i++) {
                    if (nearFloor(location, i)) {
                        return true;
                    }
                }

                return surfaceZone(
                    location,
                    homeOrigin
                );

            case DEPOSITING_LOOT:
                return location.distanceTo(ANCHOR) <= 15;

            case COMPLETED:
            case FAILED:
                return true;

            default:
                return false;
        }
    }

    private static boolean surfaceZone(
        WorldPoint location,
        WorldPoint origin
    ) {
        int minX = Math.min(
            ENTRANCE_STAND.getX(),
            origin.getX()
        );

        int minY = Math.min(
            ENTRANCE_STAND.getY(),
            origin.getY()
        );

        int maxX = Math.max(
            ANCHOR.getX(),
            origin.getX()
        );

        int maxY = Math.max(
            ANCHOR.getY(),
            origin.getY()
        );

        return location.getPlane() == 0
            && location.getX() >= minX - 15
            && location.getX() <= maxX + 15
            && location.getY() >= minY - 15
            && location.getY() <= maxY + 15;
    }

    private static int floor(WorldPoint location) {
        for (int i = 0; i < FLOORS.length; i++) {
            int[] rectangle = FLOORS[i];

            if (location.getPlane() == 0
                && location.getX() >= rectangle[0]
                && location.getX() <= rectangle[2]
                && location.getY() >= rectangle[1]
                && location.getY() <= rectangle[3]) {
                return i;
            }
        }

        return -1;
    }

    private static boolean nearFloor(
        WorldPoint location,
        int floor
    ) {
        int[] rectangle = FLOORS[floor];

        return location.getPlane() == 0
            && location.getX() >= rectangle[0] - 15
            && location.getX() <= rectangle[2] + 15
            && location.getY() >= rectangle[1] - 15
            && location.getY() <= rectangle[3] + 15;
    }

    private long stateBudget() {
        if (state == State.HOMING_TO_BANK) {
            return 900000;
        }

        if (state == State.DEPOSITING_LOOT
            || state == State.INTERACTING_ENTRANCE) {
            return 120000;
        }

        return 720000;
    }

    private void abort(
        String reason,
        WorldPoint location
    ) {
        if (!canAct() || terminal()) {
            return;
        }

        if (failure == null) {
            failure = reason;
        }

        if (location != null
            && floor(location) < 0
            && location.getPlane() == 0) {
            homeOrigin = location;
        }

        /*
         * Do not restart the homing deadline on every recovery check.
         */
        if (state != State.HOMING_TO_BANK) {
            transition(
                State.HOMING_TO_BANK,
                "ABORT: " + reason
            );
        }
    }

    private void transition(
        State next,
        String reason
    ) {
        if (!canAct() || state == next) {
            return;
        }

        State previous = state;
        state = next;

        stateSince = now();
        lastMovement = stateSince;
        stuckRetries = 0;

        objectAttempts = 0;
        lastObjectClick = 0;

        rewardSince = 0;
        lastRewardClick = 0;

        clearRoute();

        Microbot.log(
            "[Stronghold] "
                + previous
                + " -> "
                + next
                + " | "
                + reason
        );

        save(reason);
    }

    private void finish(
        State terminalState,
        String reason
    ) {
        if (!canAct()) {
            return;
        }

        if (terminalState != State.COMPLETED
            && terminalState != State.FAILED) {
            throw new IllegalArgumentException(
                "finish requires a terminal state"
            );
        }

        transition(
            terminalState,
            reason
        );

        writeJournal(
            terminalState.name(),
            reason
        );

        active = false;

        if (ownsWalkerSettings) {
            Rs2Walker.disableTeleports = previousDisableTeleports;
            ownsWalkerSettings = false;
        }

        if (mainScheduledFuture != null) {
            mainScheduledFuture.cancel(false);
        }

        super.shutdown();

        Microbot.log(
            "[Stronghold] EXIT "
                + terminalState
                + " | "
                + reason
        );
    }

    private void save(String detail) {
        if (canAct()) {
            writeJournal(
                terminal() ? state.name() : "RUNNING",
                detail
            );
        }
    }

    private void writeJournal(
        String status,
        String detail
    ) {
        if (accountKey == null) {
            return;
        }

        configManager.setConfiguration(
            StrongholdConfig.GROUP,
            accountKey,
            status
                + "|"
                + state
                + "|"
                + Instant.now()
                + "|receipts="
                + Arrays.toString(receipts)
                + "|boots="
                + bootsReceived
                + "|"
                + detail
        );
    }

    private void clearRoute() {
        goal = null;
        stepGoal = null;
        route = Collections.emptyList();
        pendingBarrier = null;

        if (ownsWalkerSettings) {
            Rs2Walker.setTarget(
                null,
                "stage0:route-reset"
            );
        }
    }

    private boolean allCoinReceipts() {
        return receipts[0]
            && receipts[1]
            && receipts[2];
    }

    private boolean terminal() {
        return state == State.COMPLETED
            || state == State.FAILED;
    }

    private boolean canAct() {
        return active
            && !Thread.currentThread().isInterrupted();
    }

    private static void actionDelay() {
        Global.sleep(Rs2Random.between(1200, 2800));
    }

    private static long now() {
        return TimeUnit.NANOSECONDS.toMillis(
            System.nanoTime()
        );
    }

    private static WorldPoint point(
        int x,
        int y
    ) {
        return new WorldPoint(x, y, 0);
    }

    private static int nearestIndex(
        List<WorldPoint> path,
        WorldPoint location
    ) {
        int best = -1;
        int distance = Integer.MAX_VALUE;

        for (int i = 0; i < path.size(); i++) {
            int candidate = location.distanceTo(
                path.get(i)
            );

            if (candidate < distance) {
                distance = candidate;
                best = i;
            }
        }

        return best;
    }

    private static String normalize(String text) {
        if (text == null) {
            return "";
        }

        return text
            .replaceAll("(?i)<br\\s*/?>", " ")
            .replaceAll("<[^>]*>", "")
            .replace('\u00a0', ' ')
            .replace('\u2019', '\'')
            .replace("&nbsp;", " ")
            .replace("&apos;", "'")
            .replaceAll("\\s+", " ")
            .trim()
            .toLowerCase(Locale.ROOT);
    }

    private static Map<String, String> buildAnswers() {
        Map<String, String> result = new LinkedHashMap<>();

        /*
         * The pinned upstream enum contains 37 question/answer pairs, including
         * phishing, account sharing, passwords, PINs, authentication, recovery,
         * compromised devices, fake moderators, streams, and giveaway scams.
         *
         * Each normalized question fragment is a substring key. Using upstream
         * wording avoids mismatches caused by paraphrased answer text.
         */
        for (StrongholdAnswer entry : StrongholdAnswer.values()) {
            String previous = result.put(
                normalize(entry.getQuestion()),
                entry.getAnswer()
            );

            if (previous != null
                && !previous.equals(entry.getAnswer())) {
                throw new IllegalStateException(
                    "Conflicting upstream Stronghold answer entries"
                );
            }
        }

        if (result.size() < 20) {
            throw new IllegalStateException(
                "Incomplete upstream Stronghold question table"
            );
        }

        return Collections.unmodifiableMap(result);
    }

    private static Snapshot snapshot() {
        return Microbot.getClientThread()
            .runOnClientThreadOptional(() -> {
                if (Microbot.getClient().getGameState()
                    != GameState.LOGGED_IN) {
                    return null;
                }

                Player player =
                    Microbot.getClient().getLocalPlayer();

                ItemContainer inventory =
                    Microbot.getClient().getItemContainer(
                        InventoryID.INVENTORY
                    );

                if (player == null
                    || player.getName() == null
                    || inventory == null) {
                    return null;
                }

                Snapshot result = new Snapshot();

                result.name = normalize(
                    player.getName()
                );

                result.location =
                    player.getWorldLocation();

                result.hp =
                    Microbot.getClient().getBoostedSkillLevel(
                        Skill.HITPOINTS
                    );

                result.maxHp =
                    Microbot.getClient().getRealSkillLevel(
                        Skill.HITPOINTS
                    );

                for (int i = 0; i < EMOTES.length; i++) {
                    result.emotes[i] =
                        Microbot.getClient().getVarbitValue(
                            EMOTES[i]
                        ) != 0;
                }

                for (Item item : inventory.getItems()) {
                    if (item.getId() < 0 || item.getQuantity() <= 0) {
                        continue;
                    }

                    result.usedSlots++;

                    if (item.getId() == ItemID.COINS) {
                        result.coins += item.getQuantity();
                    }

                    if (item.getId() == ItemID.SOS_BOOTS) {
                        result.boots += item.getQuantity();
                    }
                }

                ItemContainer equipment =
                    Microbot.getClient().getItemContainer(
                        InventoryID.EQUIPMENT
                    );

                if (equipment != null) {
                    for (Item item : equipment.getItems()) {
                        if (item.getId() >= 0
                            && item.getQuantity() > 0) {
                            result.equipmentSlots++;
                        }
                    }
                }

                return result;
            })
            .orElse(null);
    }

    private static final class Snapshot {
        private String name;
        private WorldPoint location;

        private int hp;
        private int maxHp;

        private int coins;
        private int boots;

        private int usedSlots;
        private int equipmentSlots;

        private final boolean[] emotes = new boolean[4];
    }

    private static final class Barrier {
        private final int index;
        private final WorldPoint from;
        private final WorldPoint to;
        private final WorldPoint location;

        private final int id;
        private final String action;

    private void cycle() {
        if (!active || !inCycle.compareAndSet(false, true)) {
            return;
        }

        try {
            if (!super.run() || !active) {
                return;
            }

            if (!Microbot.isLoggedIn()) {
                if (offlineSince == 0) {
                    offlineSince = now();
                } else if (now() - offlineSince > 90000) {
                    finish(State.FAILED, "Disconnected for more than 90 seconds.");
                }
                return;
            }

            offlineSince = 0;

            Snapshot snapshot = snapshot();
            if (snapshot == null) {
                return;
            }

            if (!account.isEmpty() && !account.equalsIgnoreCase(snapshot.name)) {
                finish(State.FAILED, "Account changed while the pipeline was active.");
                return;
            }

            account = snapshot.name;

            if (snapshot.hp <= 0) {
                finish(State.FAILED, "Death detected; rewards are not certified.");
                return;
            }

            // First closed-loop check on every eligible iteration, before any input.
            if (config.enableCrashRecovery() && !expectedZone(snapshot.tile)) {
                if (state == State.HOMING_TO_BANK) {
                    finish(State.FAILED, "Moved outside the recovery route.");
                } else {
                    recover("Player left the expected state zone.", snapshot.tile);
                }
                return;
            }

            if (now() - startedAt > TimeUnit.MINUTES.toMillis(40)) {
                finish(State.FAILED, "Overall 40-minute deadline exceeded.");
                return;
            }

            if (now() - enteredAt > stateBudget()) {
                if (state == State.HOMING_TO_BANK || state == State.DEPOSITING_LOOT) {
                    finish(State.FAILED, "Recovery/banking deadline exceeded.");
                } else {
                    recover("State deadline exceeded: " + state, snapshot.tile);
                }
                return;
            }

            // Never overlap a cancelled blocking walk with manual inputs.
            if (surfaceTask != null && !surfaceTask.isDone()) {
                monitorSurface(snapshot.tile);
                return;
            }

            if (!surfaceIdle.get()) {
                return;
            }

            if (surfaceError != null) {
                String error = surfaceError;
                surfaceError = null;

                if (state == State.HOMING_TO_BANK) {
                    finish(State.FAILED, "Recovery walker failed: " + error);
                } else {
                    recover("Surface walker failed: " + error, snapshot.tile);
                }
                return;
            }

            if (snapshot.hp * 100 / Math.max(1, snapshot.maxHp) <= 75
                    && (floor(snapshot.tile) >= 0
                    || state == State.WALKING_TO_STRONGHOLD)) {
                if (Rs2Player.useFood()) {
                    interactionDelay();
                    return;
                } else if (state != State.HOMING_TO_BANK) {
                    recover("Low health with no usable food.", snapshot.tile);
                    return;
                }

                // Without food, continue evacuation instead of idling in combat.
            }

            if (Rs2Dialogue.isInDialogue()) {
                handleDialogue(snapshot);
                return;
            }

            switch (state) {
                case VALIDATING_ANCHOR:
                    validate(snapshot);
                    break;

                case WALKING_TO_STRONGHOLD:
                    if (near(snapshot.tile, ENTRANCE_STAND, 4)) {
                        transition(State.INTERACTING_ENTRANCE, "Entrance reached.");
                    } else {
                        navigate(ENTRANCE_STAND, snapshot.tile, 3);
                    }
                    break;

                case INTERACTING_ENTRANCE:
                    if (floor(snapshot.tile) == 0) {
                        transition(
                            State.TRAVERSING_FLOOR_1,
                            "Observed entry into Vault of War."
                        );
                    } else {
                        interactAt(
                            ObjectID.SOS_DUNG_ENT_OPEN,
                            "Climb-down",
                            ENTRANCE,
                            ENTRANCE_STAND,
                            snapshot.tile
                        );
                    }
                    break;

                case TRAVERSING_FLOOR_1:
                    traverse(0, snapshot);
                    break;

                case TRAVERSING_FLOOR_2:
                    traverse(1, snapshot);
                    break;

                case TRAVERSING_FLOOR_3:
                    traverse(2, snapshot);
                    break;

                case TRAVERSING_FLOOR_4:
                    if (navigate(REWARD_STANDS[3], snapshot.tile, 1)) {
                        transition(State.CLAIMING_REWARDS, "Cradle room reached.");
                    }
                    break;

                case CLAIMING_REWARDS:
                    claimBoots(snapshot);
                    break;

                case HOMING_TO_BANK:
                    home(snapshot);
                    break;

                case DEPOSITING_LOOT:
                    deposit(snapshot);
                    break;

                case COMPLETED:
                case FAILED:
                    active = false;
                    break;

                default:
                    throw new IllegalStateException("Unhandled state: " + state);
            }
        } catch (RuntimeException exception) {
            // An unexpected API failure is not evidence that an input succeeded.
            finish(
                State.FAILED,
                exception.getClass().getSimpleName() + ": " + exception.getMessage()
            );
        } finally {
            if (active && !Thread.currentThread().isInterrupted()) {
                interactionDelay();
            }
            inCycle.set(false);
        }
    }

    private void validate(Snapshot snapshot) {
        if (!near(snapshot.tile, ANCHOR, 15)) {
            recover("Start location is not the GE anchor.", snapshot.tile);
            return;

        for (int index = 0; index < 3; index++) {
            if (snapshot.claimed[index]) {
                finish(
                    State.FAILED,
                    "A coin reward was already claimed. This stage requires a fresh reward run."
                );
                return;

        if (snapshot.claimed[3] || snapshot.boots > 0) {
            finish(
                State.FAILED,
                "Stronghold completion already exists; refusing to certify another 10,000 GP."
            );
        } else if (Rs2Inventory.emptySlotCount() < 2) {
            finish(State.FAILED, "Keep two inventory slots free for coins and boots.");
        } else if (Rs2Inventory.getInventoryFood().size() < 4) {
            finish(
                State.FAILED,
                "Bring at least four recognized food items; the lower floors are dangerous."
            );
        } else if (Rs2Bank.isOpen()) {
            Rs2Bank.closeBank();
            interactionDelay();
        } else {
            baselineCoins = snapshot.coins;
            validated = true;
            transition(
                State.WALKING_TO_STRONGHOLD,
                "Fresh rewards and supplies verified."
            );
        }
    }

            return;

                "Emote unlocked but expected coin total is missing.",
                snapshot.tile

            return;

        interactAt(
            DOWN_IDS[index],
            "Climb-down",
            DOWN_OBJECTS[index],
            DOWN_STANDS[index],
            snapshot.tile
        );
    }

    private void interactAt(
            int id,
            String action,
            WorldPoint objectAt,
            WorldPoint stand,
            WorldPoint player) {
        if (!navigate(stand, player, 1)) {
            return;
        }

        TileObject object = Rs2GameObject.<TileObject>getAll(
            candidate -> candidate.getId() == id
                && near(candidate.getWorldLocation(), objectAt, 2)
        ).stream()
            .min(
                Comparator.comparingInt(
                    (TileObject candidate) ->
                        candidate.getWorldLocation().distanceTo2D(objectAt)
                )
                    .thenComparingInt(
                        candidate -> candidate.getWorldLocation().getX()
                    )
                    .thenComparingInt(
                        candidate -> candidate.getWorldLocation().getY()
                    )
            )
            .orElse(null);

        if (object == null || !near(player, object.getWorldLocation(), 4)) {
            recordAttempt(id + ":missing", player);
            return;
        }

        if (!Rs2GameObject.hasAction(object, action)) {
            recover(
                "Expected action '" + action + "' is missing on object " + id,
                player
            );
            return;
        }

        if (!Rs2Tile.isTileObjectReachable(object)) {
            resetPlan();
            return;
        }

        if (!recordAttempt(id + ":" + action, player) || !canInput()) {
            return;
        }

        Rs2GameObject.interact(object, action);
        interactionDelay();
        resetPlan();
    }

    private boolean recordAttempt(String key, WorldPoint player) {
        if (!key.equals(actionKey)) {
            actionKey = key;
            actionAttempts = 0;
        }

        if (++actionAttempts > 12) {
            if (state == State.HOMING_TO_BANK) {
                finish(
                    State.FAILED,
                    "Repeated evacuation interaction failed: " + key
                );
            } else {
                recover(
                    "Repeated interaction failed: " + key
                        + ". Check account 2FA and inventory space.",
                    player
                );
            }
            return false;
        }

        return true;
    }

    /**
     * Surface travel uses the full webwalker on its own cancellable worker.
     * Dungeon travel uses planner snapshots plus walkStep for door-free approaches.
     * Calling full walkTo inside the dungeon would let its built-in door handler
     * consume security questions before this script can apply its reading/safety rules.
     */
    private boolean navigate(WorldPoint base, WorldPoint player, int radius) {
        base = resolveStand(base, player);

        if (near(player, base, radius)
                && (floor(player) < 0
                || Rs2Tile.getReachableTilesFromTile(
                    player,
                    radius + 3
                ).containsKey(base))) {
            cancelNavigation();
            return true;
        }

        if (Rs2Player.isAnimating() && !Rs2Player.isMoving()) {
            return false;
        }

        if (!base.equals(destination)) {
            cancelNavigation();
            destination = base;
            walkTarget = jitter(base, player, false);
            previousTile = player;
            lastMovedAt = now();
        }

        if (floor(player) < 0) {
            if (surfaceIdle.get()
                    && (surfaceTask == null || surfaceTask.isDone())) {
                final WorldPoint target = walkTarget;

                surfaceTask = walkingExecutor.submit(() -> {
                    surfaceIdle.set(false);
                    try {
                        if (canInput()) {
                            Rs2Walker.walkTo(
                                target,
                                Math.max(1, radius - 1)
                            );
                            interactionDelay();
                        }
                    } catch (RuntimeException exception) {
                        if (active && !Thread.currentThread().isInterrupted()) {
                            surfaceError = exception.toString();
                        }
                    } finally {
                        surfaceIdle.set(true);
                    }
                });
            }

            return false;
        }

        if (pendingTransport != null) {
            if (near(player, pendingTransport, 3)
                    || now() - pendingTransportAt >= 8000) {
                resetPlan();
                lastMovedAt = now();
            } else {
                return false;
            }
        }

        if (pendingDoor != null) {
            WallObject wall = (WallObject) pendingDoor.object;
            WorldPoint wallTile = wall.getWorldLocation();

            if (Rs2DoorGeometry.playerBeyondWallFace(
                    wall.getOrientationA(),
                    wallTile,
                    pendingDoorFrom,
                    player
                )
                    || Rs2DoorGeometry.playerBeyondWallFace(
                        wall.getOrientationB(),
                        wallTile,
                        pendingDoorFrom,
                        player
                    )) {
                resetPlan();
                lastMovedAt = now();
            } else if (now() - pendingDoorAt < 8000 || Rs2Player.isMoving()) {
                return false;
            } else {
                resetPlan();
            }
        }

        if (!player.equals(previousTile)) {
            previousTile = player;
            lastMovedAt = now();
        } else if (now() - lastMovedAt >= 8000
                && !Rs2Dialogue.isInDialogue()) {
            Microbot.log(
                "[Stronghold] No tile movement for 8 seconds; replanning dungeon walk."
            );
            resetPlan();
            lastMovedAt = now();
        }

        if (plannedPath.isEmpty()) {
            if (plannedAt == 0) {
                Rs2Walker.setTarget(walkTarget);
                plannedAt = now();
                return false;
            }

            Rs2ActiveRouteStatus status = Rs2PathApi.getActiveRouteStatus();

            if (!status.isReady()
                    || !status.getTargets().contains(walkTarget)) {
                if (now() - plannedAt > 20000) {
                    resetPlan();
                }
                return false;
            }

            List<WorldPoint> path = status.getRawPath();

            if (path.isEmpty()
                    || !near(
                        path.get(path.size() - 1),
                        walkTarget,
                        2
                    )) {
                recover(
                    "Dungeon planner did not reach the requested destination.",
                    player
                );
                return false;
            }

            plannedPath = new ArrayList<>(path);
            plannedRoute = Rs2PathApi.getActiveRoute().orElse(null);

            if (plannedRoute == null
                    || !plannedRoute.getPath().equals(plannedPath)) {
                resetPlan();
                return false;
            }
        }

        int nearest = 0;

        for (int index = 1; index < plannedPath.size(); index++) {
            if (player.distanceTo2D(plannedPath.get(index))
                    < player.distanceTo2D(plannedPath.get(nearest))) {
                nearest = index;
            }
        }

        Rs2TransportEdge transport = null;
        int transportIndex = plannedPath.size() - 1;

        for (int index = nearest; index + 1 < plannedPath.size(); index++) {
            Rs2TransportEdge edge = plannedRoute.getTransportEdge(
                plannedPath.get(index),
                plannedPath.get(index + 1)
            ).orElse(null);

            if (edge != null) {
                transport = edge;
                transportIndex = index;
                break;
            }
        }

        final List<WorldPoint> route = plannedPath;
        final int startIndex = nearest;
        final int scanEnd = transportIndex;

        Door door = Microbot.getClientThread().runOnClientThreadOptional(() -> {
            List<TileObject> objects = Rs2GameObject.<TileObject>getAll(
                candidate -> isStrongholdDoor(candidate.getId())
            );

            objects.sort(
                Comparator.comparingInt(TileObject::getId)
                    .thenComparingInt(
                        candidate -> candidate.getWorldLocation().getX()
                    )
                    .thenComparingInt(
                        candidate -> candidate.getWorldLocation().getY()
                    )
            );

            for (int index = startIndex; index < scanEnd; index++) {
                for (TileObject object : objects) {
                    if (object instanceof WallObject
                            && Rs2DoorGeometry.wallDoorTouchesSegment(
                                (WallObject) object,
                                route.get(index),
                                route.get(index + 1)
                            )) {
                        return new Door(object, route.get(index));
                    }
                }
            }

            return null;
        }).orElse(null);

        WorldPoint approach = door != null
            ? door.before
            : transport != null
                ? transport.getOrigin()
                : walkTarget;

        if (door == null
                && transport != null
                && near(player, approach, 1)
                && !Rs2Player.isMoving()) {
            if (floor(transport.getDestination()) != floor(player)
                    || !isShortcut(transport.getObjectId())) {
                recover(
                    "Planner selected an unsupported dungeon transport: "
                        + transport.getDisplayInfo(),
                    player
                );
                return false;
            }

            TileObject object = Rs2GameObject.findObjectByIdAndDistance(
                transport.getObjectId(),
                4
            );

            if (object != null
                    && Rs2GameObject.hasAction(object, transport.getAction())
                    && canInput()
                    && recordAttempt(
                        "shortcut:" + transport.getObjectId(),
                        player
                    )) {
                Rs2GameObject.interact(object, transport.getAction());
                interactionDelay();

                WorldPoint expected = transport.getDestination();

                resetPlan();
                pendingTransport = expected;
                pendingTransportAt = now();
            }

            return false;
        }

        if (door != null
                && near(player, approach, 1)
                && near(player, door.object.getWorldLocation(), 2)) {
            if (!Rs2Player.isMoving() && canInput()) {
                String action = Rs2GameObject.hasAction(door.object, "Open")
                    ? "Open"
                    : "Pass-through";

                if (!Rs2GameObject.hasAction(door.object, action)) {
                    recover("Unrecognized Stronghold door action.", player);
                    return false;
                }

                if (!recordAttempt(
                        "door:" + door.object.getId()
                            + ":" + door.object.getWorldLocation(),
                        player)) {
                    return false;
                }

                Rs2GameObject.interact(door.object, action);
                interactionDelay();

                resetPlan();
                pendingDoor = door;
                pendingDoorFrom = player;
                pendingDoorAt = now();
            }
        } else {
            if (stepTarget == null || !near(stepTarget, approach, 1)) {
                stepTarget = jitter(approach, player, true);
            }

            WalkerState result = Rs2Walker.walkStep(stepTarget, 0);

            if (result == WalkerState.UNREACHABLE
                    || result == WalkerState.EXIT) {
                resetPlan();
            }
        }

        return false;
    }

    private void monitorSurface(WorldPoint player) {
        if (!player.equals(previousTile)) {
            previousTile = player;
            lastMovedAt = now();
        } else if (now() - lastMovedAt >= 8000) {
            Microbot.log(
                "[Stronghold] No tile movement for 8 seconds; cancelling and reissuing walkTo."
            );

            surfaceTask.cancel(true);
            Rs2Walker.setTarget(null);
            lastMovedAt = now();

            // A subsequent cycle reissues only after the old worker's finally block ran.
        }
    }

    private WorldPoint resolveStand(WorldPoint base, WorldPoint player) {
        if (floor(player) < 0
                || !near(base, player, 8)
                || Rs2Tile.isWalkable(base)) {
            return base;
        }

        List<WorldPoint> candidates = new ArrayList<>();

        for (int dx = -1; dx <= 1; dx++) {
            for (int dy = -1; dy <= 1; dy++) {
                WorldPoint candidate = new WorldPoint(
                    base.getX() + dx,
                    base.getY() + dy,
                    base.getPlane()
                );

                if (Rs2Tile.isWalkable(candidate)) {
                    candidates.add(candidate);
                }
            }
        }

        return candidates.stream()
            .min(
                Comparator.comparingInt(
                    (WorldPoint candidate) ->
                        Math.abs(candidate.getX() - base.getX())
                            + Math.abs(candidate.getY() - base.getY())
                )
                    .thenComparingInt(WorldPoint::getX)
                    .thenComparingInt(WorldPoint::getY)
            )
            .orElse(base);
    }

    private WorldPoint jitter(
            WorldPoint base,
            WorldPoint player,
            boolean requireReachable) {
        WorldPoint candidate = config.enableJitter()
            ? new WorldPoint(
                base.getX() + Rs2Random.between(-1, 2),
                base.getY() + Rs2Random.between(-1, 2),
                base.getPlane()
            )
            : base;

        if (floor(player) < 0) {
            return candidate;
        }

        boolean walkable = Rs2Tile.isWalkable(candidate);

        if (walkable
                && Rs2Tile.getReachableTilesFromTile(
                    base,
                    4
                ).containsKey(candidate)
                && (!requireReachable
                || Rs2Tile.getReachableTilesFromTile(
                    player,
                    64
                ).containsKey(candidate))) {
            return candidate;
        }

        return base;
    }

    private void handleDialogue(Snapshot snapshot) {
        if (state == State.HOMING_TO_BANK && !failure.isEmpty()) {
            if (canInput() && Rs2Dialogue.isInDialogue()) {
                Rs2Keyboard.keyPress(KeyEvent.VK_ESCAPE);
                interactionDelay();
            }
            return;
        }

        String text = normalize(Rs2Dialogue.getDialogueText());
        String question = normalize(Rs2Dialogue.getQuestion());

        if (!text.isEmpty()) {
            lastDialogue = text;
        }

        String answer = findAnswer(text + " " + question);

        if (answer != null) {
            rememberedAnswer = answer;
        }

        if (Rs2Dialogue.hasSelectAnOption()) {
            List<String> options = optionTexts();
            String choice = null;

            if (state == State.CLAIMING_REWARDS) {
                choice = options.stream()
                    .filter(
                        option -> normalize(option).matches(
                            ".*\\bfancy boots\\b.*"
                        )
                    )
                    .findFirst()
                    .orElse(null);
            }

            if (choice == null
                    && state == State.CLAIMING_REWARDS
                    && fancySelected
                    && (lastDialogue.contains("sure")
                    || lastDialogue.contains("fancy boots"))) {
                choice = options.stream()
                    .filter(
                        option -> normalize(option).startsWith("yes")
                    )
                    .findFirst()
                    .orElse(null);
            }

            if (choice == null && rememberedAnswer != null) {
                final String expected = normalize(rememberedAnswer);

                choice = options.stream()
                    .filter(
                        option -> normalize(option).equals(expected)
                    )
                    .findFirst()
                    .orElse(null);
            }

            if (choice == null
                    && descentPending
                    && (lastDialogue.contains("danger")
                    || lastDialogue.contains("next level")
                    || lastDialogue.contains("descend"))) {
                choice = options.stream()
                    .filter(
                        option -> normalize(option).startsWith("yes")
                    )
                    .findFirst()
                    .orElse(null);
            }

            if (choice == null) {
                finish(
                    State.FAILED,
                    "Unknown dialogue; no answer guessed. Question="
                        + question
                        + "; context=" + lastDialogue
                        + "; options=" + options
                );
                return;
            }

            final String selected = choice;
            final List<String> before = options;

            readingDelay();

            if (canInput()
                    && Rs2Dialogue.isInDialogue()
                    && Rs2Dialogue.hasSelectAnOption()
                    && before.equals(optionTexts())
                    && question.equals(
                        normalize(Rs2Dialogue.getQuestion())
                    )) {
                boolean clicked = Rs2Dialogue.clickOption(selected, true);

                if (clicked
                        && state == State.CLAIMING_REWARDS
                        && normalize(selected).matches(
                            ".*\\bfancy boots\\b.*"
                        )) {
                    fancySelected = true;
                }

                if (clicked) {
                    rememberedAnswer = null;
                }

                pendingDoorAt = now();
                interactionDelay();
            }
        } else if (Rs2Dialogue.hasContinue()) {
            readingDelay();

            if (canInput()
                    && Rs2Dialogue.isInDialogue()
                    && Rs2Dialogue.hasContinue()
                    && text.equals(
                        normalize(Rs2Dialogue.getDialogueText())
                    )) {
                Rs2Dialogue.clickContinue();
                pendingDoorAt = now();
                interactionDelay();
            }
        }
    }

    private List<String> optionTexts() {
        return Microbot.getClientThread().runOnClientThreadOptional(() -> {
            List<String> result = new ArrayList<>();

            for (Widget option : Rs2Dialogue.getDialogueOptions()) {
                result.add(option.getText());
            }

            return result;
        }).orElse(Collections.emptyList());
    }

    private static String findAnswer(String question) {
        for (Map.Entry<String, String> entry : SECURITY_ANSWERS.entrySet()) {
            if (question.contains(entry.getKey())) {
                return entry.getValue();
            }
        }

        return null;
    }

    private boolean expectedZone(WorldPoint tile) {
        switch (state) {
            case VALIDATING_ANCHOR:
                return near(tile, ANCHOR, 15);

            case WALKING_TO_STRONGHOLD:
                return inRectangle(
                    tile,
                    3060,
                    3400,
                    3190,
                    3510,
                    15
                );

            case INTERACTING_ENTRANCE:
                return near(tile, ENTRANCE, 15)
                    || inFloor(tile, 0, 15);

            case TRAVERSING_FLOOR_1:
            case TRAVERSING_FLOOR_2:
            case TRAVERSING_FLOOR_3:
            case TRAVERSING_FLOOR_4:
                int index = state.ordinal()
                    - State.TRAVERSING_FLOOR_1.ordinal();

                return inFloor(tile, index, 15)
                    || (descentPending
                    && index < 3
                    && inFloor(tile, index + 1, 15));

            case CLAIMING_REWARDS:
                return inFloor(tile, 3, 15);

            case HOMING_TO_BANK:
                for (int floorIndex = 0; floorIndex < 4; floorIndex++) {
                    if (inFloor(tile, floorIndex, 15)) {
                        return true;
                    }
                }

                int x = recoveryOrigin != null
                    && floor(recoveryOrigin) < 0
                        ? recoveryOrigin.getX()
                        : 3081;

                int y = recoveryOrigin != null
                    && floor(recoveryOrigin) < 0
                        ? recoveryOrigin.getY()
                        : 3421;

                return inRectangle(
                    tile,
                    Math.min(x, 3060) - 32,
                    Math.min(y, 3400) - 32,
                    Math.max(x, 3190) + 32,
                    Math.max(y, 3510) + 32,
                    15
                );

            case DEPOSITING_LOOT:
                return near(tile, ANCHOR, 15);

            case COMPLETED:
            case FAILED:
                return true;

            default:
                return false;
        }
    }

    private void recover(String reason, WorldPoint location) {
        if (failure.isEmpty()) {
            failure = reason;
        }

        rewardsVerified = false;
        recoveryOrigin = location;

        transition(State.HOMING_TO_BANK, "Recovery: " + reason);
    }

    private void transition(State next, String reason) {
        State previous = state;

        cancelNavigation();

        state = next;
        enteredAt = now();
        actionKey = "";
        actionAttempts = 0;
        descentPending = false;
        rememberedAnswer = null;
        lastDialogue = "";

        Microbot.log(
            "[Stronghold] " + previous + " -> " + next + " | " + reason
        );

        journal(reason);
    }

    private void finish(State terminal, String reason) {
        if (!active) {
            return;
        }

        if (terminal == State.FAILED && failure.isEmpty()) {
            failure = reason;
        }

        transition(terminal, reason);
        shutdown();
    }

    private void journal(String reason) {
        if (!account.isEmpty()) {
            String key = "checkpoint_"
                + account.toLowerCase(Locale.ROOT)
                    .replaceAll("[^a-z0-9]", "_");

            try {
                configManager.setConfiguration(
                    StrongholdConfig.GROUP,
                    key,
                    Instant.now()
                        + " | account=" + account
                        + " | state=" + state
                        + " | verified=" + rewardsVerified
                        + " | baselineCoins=" + baselineCoins
                        + " | reason=" + reason.replace('\n', ' ')
                );
            } catch (RuntimeException exception) {
                Microbot.log(
                    "[Stronghold] Checkpoint persistence failed: "
                        + exception
                );
            }
        }
    }

    private void resetPlan() {
        Rs2Walker.setTarget(null);
        plannedPath = Collections.emptyList();
        plannedAt = 0;
        stepTarget = null;
        pendingDoor = null;
        pendingTransport = null;
        plannedRoute = null;
    }

    private void cancelNavigation() {
        if (surfaceTask != null && !surfaceTask.isDone()) {
            surfaceTask.cancel(true);
        }

        resetPlan();

        destination = null;
        walkTarget = null;
    }

    @Override
    public synchronized void shutdown() {
        if (active && state != State.COMPLETED && state != State.FAILED) {
            State previous = state;
            state = State.FAILED;
            failure = "Plugin stopped before completion.";

            Microbot.log(
                "[Stronghold] " + previous + " -> FAILED | " + failure
            );

            journal(failure);
        }

        active = false;
        cancelNavigation();

        if (walkingExecutor != null) {
            walkingExecutor.shutdownNow();
        }

        super.shutdown();

        if (ownsWalkerSettings) {
            Rs2Walker.disableTeleports = oldDisableTeleports;
            ownsWalkerSettings = false;
        }
    }

    private boolean canInput() {
        return active
            && !Thread.currentThread().isInterrupted()
            && Microbot.isLoggedIn()
            && !Microbot.pauseAllScripts.get()
            && !InputArbiter.isHuman();
    }

    private long stateBudget() {
        switch (state) {
            case TRAVERSING_FLOOR_1:
            case TRAVERSING_FLOOR_2:
            case TRAVERSING_FLOOR_3:
            case TRAVERSING_FLOOR_4:
            case HOMING_TO_BANK:
                return TimeUnit.MINUTES.toMillis(10);

            default:
                return TimeUnit.MINUTES.toMillis(2);
        }
    }

    private Snapshot snapshot() {
        return Microbot.getClientThread().runOnClientThreadOptional(() -> {
            if (Microbot.getClient().getLocalPlayer() == null) {
                return null;
            }

            Snapshot result = new Snapshot();

            result.tile = Microbot.getClient()
                .getLocalPlayer()
                .getWorldLocation();

            result.name = Microbot.getClient()
                .getLocalPlayer()
                .getName();

            if (result.name == null) {
                return null;
            }

            result.hp = Microbot.getClient()
                .getBoostedSkillLevel(Skill.HITPOINTS);

            result.maxHp = Microbot.getClient()
                .getRealSkillLevel(Skill.HITPOINTS);

            for (int index = 0; index < REWARD_BITS.length; index++) {
                result.claimed[index] = Microbot.getClient()
                    .getVarbitValue(REWARD_BITS[index]) != 0;
            }

            ItemContainer inventory = Microbot.getClient()
                .getItemContainer(InventoryID.INVENTORY);

            ItemContainer equipment = Microbot.getClient()
                .getItemContainer(InventoryID.EQUIPMENT);

            if (inventory == null || equipment == null) {
                return null;
            }

            result.inventoryEmpty = empty(inventory);
            result.equipmentEmpty = empty(equipment);

            for (Item item : inventory.getItems()) {
                if (item.getId() == COINS) {
                    result.coins += item.getQuantity();
                } else if (item.getId() == FANCY_BOOTS) {
                    result.boots += item.getQuantity();
                }
            }

            return result;
        }).orElse(null);
    }

    private static boolean empty(ItemContainer container) {
        for (Item item : container.getItems()) {
            if (item.getId() >= 0 && item.getQuantity() > 0) {
                return false;
            }
        }

        return true;
    }

    private static boolean isShortcut(int id) {
        return id == ObjectID.SOS_WAR_PORTAL
            || id == ObjectID.SOS_FAM_PORTAL
            || id == ObjectID.SOS_PEST_PORTAL
            || id == ObjectID.SOS_DEATH_PORTAL
            || id == ObjectID.SOS_FAM_ROPE_UP
            || id == ObjectID.SOS_PEST_ROPE_UP
            || id == 1734;
    }

    private static boolean isStrongholdDoor(int id) {
        return id == ObjectID.SOS_WAR_DOOR_FACE
            || id == ObjectID.SOS_WAR_DOOR_FACE_MIRR
            || id == ObjectID.SOS_FAM_DOOR_FACE
            || id == ObjectID.SOS_FAM_DOOR_FACE_MIRR
            || id == ObjectID.SOS_PEST_DOOR_FACE
            || id == ObjectID.SOS_PEST_DOOR_FACE_MIRR
            || id == ObjectID.SOS_DEATH_DOOR_FACE
            || id == ObjectID.SOS_DEATH_DOOR_FACE_MIRR;
    }

    private static WorldPoint point(int x, int y) {
        return new WorldPoint(x, y, 0);
    }

    private static boolean near(
            WorldPoint first,
            WorldPoint second,
            int radius) {
        return first != null
            && second != null
            && first.getPlane() == second.getPlane()
            && first.distanceTo2D(second) <= radius;
    }

    private static boolean inRectangle(
            WorldPoint point,
            int minX,
            int minY,
            int maxX,
            int maxY,
            int margin) {
        return point != null
            && point.getPlane() == 0
            && point.getX() >= minX - margin
            && point.getX() <= maxX + margin
            && point.getY() >= minY - margin
            && point.getY() <= maxY + margin;
    }

    private static boolean inFloor(
            WorldPoint point,
            int index,
            int margin) {
        int[] bounds = FLOOR_BOUNDS[index];

        return inRectangle(
            point,
            bounds[0],
            bounds[1],
            bounds[2],
            bounds[3],
            margin
        );
    }

    private static int floor(WorldPoint point) {
        for (int index = 0; index < FLOOR_BOUNDS.length; index++) {
            if (inFloor(point, index, 0)) {
                return index;
            }
        }

        return -1;
    }

    private static long now() {
        return TimeUnit.NANOSECONDS.toMillis(System.nanoTime());
    }

    private static String normalize(String value) {
        return value == null
            ? ""
            : value.replaceAll("(?i)<br\\s*/?>", " ")
                .replaceAll("<[^>]*>", "")
                .replace('\u00a0', ' ')
                .replace('\u2019', '\'')
                .replaceAll("\\s+", " ")
                .trim()
                .toLowerCase(Locale.ROOT);
    }

    private static void interactionDelay() {
        // Global.sleep is inherited by Script; Microbot.sleep does not exist in this revision.
        sleep(Rs2Random.between(1200, 2800));
    }

    private static void readingDelay() {
        sleep(Rs2Random.between(1800, 4200));
    }

    private static final class Snapshot {
        private WorldPoint tile;
        private String name;
        private int hp;
        private int maxHp;
        private long coins;
        private int boots;
        private boolean inventoryEmpty;
        private boolean equipmentEmpty;
        private final boolean[] claimed = new boolean[4];
    }

    private static final class Door {
        private final TileObject object;
        private final WorldPoint before;

        private Door(TileObject object, WorldPoint before) {
            this.object = object;
            this.before = before;
        }
    }

    // Question/answer data adapted from the pinned upstream StrongholdAnswer enum.

    private void cycle() {
        if (!active || !inCycle.compareAndSet(false, true)) {
            return;
        }
        try {
            if (!super.run() || !active) {
                return;
            }
            if (!Microbot.isLoggedIn()) {
                if (offlineSince == 0) {
                    offlineSince = now();
                } else if (now() - offlineSince > 90000) {
                    finish(State.FAILED, "Disconnected for more than 90 seconds.");
                }
                return;
            }
            offlineSince = 0;
            Snapshot snapshot = snapshot();
            if (snapshot == null) {
                return;
            }
            if (!account.isEmpty() && !account.equalsIgnoreCase(snapshot.name)) {
                finish(State.FAILED, "Account changed while the pipeline was active.");
                return;
            }
            account = snapshot.name;
            if (snapshot.hp <= 0) {
                finish(State.FAILED, "Death detected; rewards are not certified.");
                return;
            }
            // First closed-loop check on every eligible iteration, before any input.
            if (config.enableCrashRecovery() && !expectedZone(snapshot.tile)) {
                if (state == State.HOMING_TO_BANK) {
                    finish(State.FAILED, "Moved outside the recovery route.");
                } else {
                    recover("Player left the expected state zone.", snapshot.tile);
                }
                return;
            }
            if (now() - startedAt > TimeUnit.MINUTES.toMillis(40)) {
                finish(State.FAILED, "Overall 40-minute deadline exceeded.");
                return;
            }
            if (now() - enteredAt > stateBudget()) {
                if (state == State.HOMING_TO_BANK || state == State.DEPOSITING_LOOT) {
                    finish(State.FAILED, "Recovery/banking deadline exceeded.");
                } else {
                    recover("State deadline exceeded: " + state, snapshot.tile);
                }
                return;
            }
            // Never overlap a cancelled blocking walk with manual inputs.
            if (surfaceTask != null && !surfaceTask.isDone()) {
                monitorSurface(snapshot.tile);
                return;
            }
            if (!surfaceIdle.get()) {
                return;
            }
            if (surfaceError != null) {
                String error = surfaceError;
                surfaceError = null;
                if (state == State.HOMING_TO_BANK) {
                    finish(State.FAILED, "Recovery walker failed: " + error);
                } else {
                    recover("Surface walker failed: " + error, snapshot.tile);
                }
                return;
            }
            if (snapshot.hp * 100 / Math.max(1, snapshot.maxHp) <= 75
                    && (floor(snapshot.tile) >= 0 || state == State.WALKING_TO_STRONGHOLD)) {
                if (Rs2Player.useFood()) {
                    interactionDelay();
                    return;
                } else if (state != State.HOMING_TO_BANK) {
                    recover("Low health with no usable food.", snapshot.tile);
                    return;
                }
                // Without food, continue evacuation instead of idling in combat.
            }
            if (Rs2Dialogue.isInDialogue()) {
                handleDialogue(snapshot);
                return;
            }
            switch (state) {
                case VALIDATING_ANCHOR:
                    validate(snapshot);
                    break;
                case WALKING_TO_STRONGHOLD:
                    if (near(snapshot.tile, ENTRANCE_STAND, 4)) {
                        transition(State.INTERACTING_ENTRANCE, "Entrance reached.");
                    } else {
                        navigate(ENTRANCE_STAND, snapshot.tile, 3);
                    }
                    break;
                case INTERACTING_ENTRANCE:
                    if (floor(snapshot.tile) == 0) {
                        transition(State.TRAVERSING_FLOOR_1, "Observed entry into Vault of War.");
                    } else {
                        interactAt(ObjectID.SOS_DUNG_ENT_OPEN, "Climb-down", ENTRANCE,
                            ENTRANCE_STAND, snapshot.tile);
                    }
                    break;
                case TRAVERSING_FLOOR_1:
                    traverse(0, snapshot);
                    break;
                case TRAVERSING_FLOOR_2:
                    traverse(1, snapshot);
                    break;
                case TRAVERSING_FLOOR_3:
                    traverse(2, snapshot);
                    break;
                case TRAVERSING_FLOOR_4:
                    if (navigate(REWARD_STANDS[3], snapshot.tile, 1)) {
                        transition(State.CLAIMING_REWARDS, "Cradle room reached.");
                    }
                    break;
                case CLAIMING_REWARDS:
                    claimBoots(snapshot);
                    break;
                case HOMING_TO_BANK:
                    home(snapshot);
                    break;
                case DEPOSITING_LOOT:
                    deposit(snapshot);
                    break;
                case COMPLETED:
                case FAILED:
                    active = false;
                    break;
                default:
                    throw new IllegalStateException("Unhandled state: " + state);
            }
        } catch (RuntimeException exception) {
            // An unexpected API failure is not evidence that an input succeeded.
            finish(State.FAILED, exception.getClass().getSimpleName() + ": " + exception.getMessage());
        } finally {
            if (active && !Thread.currentThread().isInterrupted()) {
                interactionDelay();
            }
            inCycle.set(false);
        }
    }

    private void validate(Snapshot snapshot) {
        if (!near(snapshot.tile, ANCHOR, 15)) {
            recover("Start location is not the GE anchor.", snapshot.tile);
            return;
        }
        for (int index = 0; index < 3; index++) {
            if (snapshot.claimed[index]) {
                finish(State.FAILED, "A coin reward was already claimed. This stage requires a fresh reward run.");
                return;
            }
        }
        if (snapshot.claimed[3] || snapshot.boots > 0) {
            finish(State.FAILED, "Stronghold completion already exists; refusing to certify another 10,000 GP.");
        } else if (Rs2Inventory.emptySlotCount() < 2) {
            finish(State.FAILED, "Keep two inventory slots free for coins and boots.");
        } else if (Rs2Inventory.getInventoryFood().size() < 4) {
            finish(State.FAILED, "Bring at least four recognized food items; the lower floors are dangerous.");
        } else if (Rs2Bank.isOpen()) {
            Rs2Bank.closeBank();
            interactionDelay();
        } else {
            baselineCoins = snapshot.coins;
            validated = true;
            transition(State.WALKING_TO_STRONGHOLD, "Fresh rewards and supplies verified.");
        }
    }

    private void traverse(int index, Snapshot snapshot) {
        int observedFloor = floor(snapshot.tile);
        if (observedFloor == index + 1 && descentPending && snapshot.claimed[index]) {
            transition(FLOOR_STATES[index + 1], "Observed floor transition.");
            return;
        }
        if (observedFloor != index) {
            recover("Unexpected floor during traversal.", snapshot.tile);
            return;
        }
        if (!snapshot.claimed[index]) {
            interactAt(REWARD_IDS[index], index == 1 ? "Search" : "Open", REWARDS[index],
                REWARD_STANDS[index], snapshot.tile);
            return;
        }
        if (snapshot.coins < baselineCoins + COIN_TOTALS[index]) {
            recover("Emote unlocked but expected coin total is missing.", snapshot.tile);
            return;
        }
        descentPending = true;
        interactAt(DOWN_IDS[index], "Climb-down", DOWN_OBJECTS[index], DOWN_STANDS[index], snapshot.tile);
    }

    private void deposit(Snapshot snapshot) {
        if (!near(snapshot.tile, ANCHOR, 10)) {
            recover("Moved away from the GE during banking.", snapshot.tile);
        } else if (!Rs2Bank.isOpen()) {
            if (bankDepositIssued && snapshot.inventoryEmpty && snapshot.equipmentEmpty) {
                finish(rewardsVerified && failure.isEmpty() ? State.COMPLETED : State.FAILED,
                    failure.isEmpty() ? "Rewards deposited; ready for the next stage." : failure);
            } else {
                Rs2Bank.openBank();
                interactionDelay();
            }
        } else if (!snapshot.inventoryEmpty) {
            Rs2Bank.depositAll();
            interactionDelay();
        } else if (!snapshot.equipmentEmpty) {
            Rs2Bank.depositEquipment();
            interactionDelay();
        } else {
            bankDepositIssued = true;
            Rs2Bank.closeBank();
            interactionDelay();
        }
    }

    private void interactAt(int id, String action, WorldPoint objectAt, WorldPoint stand, WorldPoint player) {
        if (!navigate(stand, player, 1)) {
            return;
        }
        TileObject object = Rs2GameObject.<TileObject>getAll(candidate -> candidate.getId() == id
                && near(candidate.getWorldLocation(), objectAt, 2)).stream()
            .min(Comparator.comparingInt((TileObject candidate) -> candidate.getWorldLocation().distanceTo2D(objectAt))
                .thenComparingInt(candidate -> candidate.getWorldLocation().getX())
                .thenComparingInt(candidate -> candidate.getWorldLocation().getY()))
            .orElse(null);
        if (object == null || !near(player, object.getWorldLocation(), 4)) {
            recordAttempt(id + ":missing", player);
            return;
        }
        if (!Rs2GameObject.hasAction(object, action)) {
            recover("Expected action '" + action + "' is missing on object " + id, player);
            return;
        }
        if (!Rs2Tile.isTileObjectReachable(object)) {
            resetPlan();
            return;
        }
        if (!recordAttempt(id + ":" + action, player) || !canInput()) {
            return;
        }
        Rs2GameObject.interact(object, action);
        interactionDelay();
        resetPlan();
    }

    private boolean recordAttempt(String key, WorldPoint player) {
        if (!key.equals(actionKey)) {
            actionKey = key;
            actionAttempts = 0;
        }
        if (++actionAttempts > 12) {
            if (state == State.HOMING_TO_BANK) {
                finish(State.FAILED, "Repeated evacuation interaction failed: " + key);
            } else {
                recover("Repeated interaction failed: " + key + ". Check account 2FA and inventory space.", player);
            }
            return false;
        }
        return true;
    }

    /**
     * Surface travel uses the full webwalker on its own cancellable worker.
     * Dungeon travel uses planner snapshots plus walkStep for door-free approaches.
     * Calling full walkTo inside the dungeon would let its built-in door handler
     * consume security questions before this script can apply its reading/safety rules.
     */
    private boolean navigate(WorldPoint base, WorldPoint player, int radius) {
        base = resolveStand(base, player);
        if (near(player, base, radius) && (floor(player) < 0
                || Rs2Tile.getReachableTilesFromTile(player, radius + 3).containsKey(base))) {
            cancelNavigation();
            return true;
        }
        if (Rs2Player.isAnimating() && !Rs2Player.isMoving()) {
            return false;
        }
        if (!base.equals(destination)) {
            cancelNavigation();
            destination = base;
            walkTarget = jitter(base, player, false);
            previousTile = player;
            lastMovedAt = now();
        }
        if (floor(player) < 0) {
            if (surfaceIdle.get() && (surfaceTask == null || surfaceTask.isDone())) {
                final WorldPoint target = walkTarget;
                surfaceTask = walkingExecutor.submit(() -> {
                    surfaceIdle.set(false);
                    try {
                        if (canInput()) {
                            Rs2Walker.walkTo(target, Math.max(1, radius - 1));
                            interactionDelay();
                        }
                    } catch (RuntimeException exception) {
                        if (active && !Thread.currentThread().isInterrupted()) {
                            surfaceError = exception.toString();
                        }
                    } finally {
                        surfaceIdle.set(true);
                    }
                });
            }
            return false;
        }
        if (pendingTransport != null) {
            if (near(player, pendingTransport, 3) || now() - pendingTransportAt >= 8000) {
                resetPlan();
                lastMovedAt = now();
            } else {
                return false;
            }
        }
        if (pendingDoor != null) {
            WallObject wall = (WallObject) pendingDoor.object;
            WorldPoint wallTile = wall.getWorldLocation();
            if (Rs2DoorGeometry.playerBeyondWallFace(wall.getOrientationA(), wallTile, pendingDoorFrom, player)
                    || Rs2DoorGeometry.playerBeyondWallFace(wall.getOrientationB(), wallTile, pendingDoorFrom, player)) {
                resetPlan();
                lastMovedAt = now();
            } else if (now() - pendingDoorAt < 8000 || Rs2Player.isMoving()) {
                return false;
            } else {
                resetPlan();
            }
        }
        if (!player.equals(previousTile)) {
            previousTile = player;
            lastMovedAt = now();
        } else if (now() - lastMovedAt >= 8000 && !Rs2Dialogue.isInDialogue()) {
            Microbot.log("[Stronghold] No tile movement for 8 seconds; replanning dungeon walk.");
            resetPlan();
            lastMovedAt = now();
        }
        if (plannedPath.isEmpty()) {
            if (plannedAt == 0) {
                Rs2Walker.setTarget(walkTarget);
                plannedAt = now();
                return false;
            }
            Rs2ActiveRouteStatus status = Rs2PathApi.getActiveRouteStatus();
            if (!status.isReady() || !status.getTargets().contains(walkTarget)) {
                if (now() - plannedAt > 20000) {
                    resetPlan();
                }
                return false;
            }
            List<WorldPoint> path = status.getRawPath();
            if (path.isEmpty() || !near(path.get(path.size() - 1), walkTarget, 2)) {
                recover("Dungeon planner did not reach the requested destination.", player);
                return false;
            }
            plannedPath = new ArrayList<>(path);
            plannedRoute = Rs2PathApi.getActiveRoute().orElse(null);
            if (plannedRoute == null || !plannedRoute.getPath().equals(plannedPath)) {
                resetPlan();
                return false;
            }
        }
        int nearest = 0;
        for (int index = 1; index < plannedPath.size(); index++) {
            if (player.distanceTo2D(plannedPath.get(index)) < player.distanceTo2D(plannedPath.get(nearest))) {
                nearest = index;
            }
        }
        Rs2TransportEdge transport = null;
        int transportIndex = plannedPath.size() - 1;
        for (int index = nearest; index + 1 < plannedPath.size(); index++) {
            Rs2TransportEdge edge = plannedRoute.getTransportEdge(plannedPath.get(index), plannedPath.get(index + 1))
                .orElse(null);
            if (edge != null) {
                transport = edge;
                transportIndex = index;
                break;
            }
        }
        final List<WorldPoint> route = plannedPath;
        final int startIndex = nearest;
        final int scanEnd = transportIndex;
        Door door = Microbot.getClientThread().runOnClientThreadOptional(() -> {
            List<TileObject> objects = Rs2GameObject.<TileObject>getAll(candidate -> isStrongholdDoor(candidate.getId()));
            objects.sort(Comparator.comparingInt(TileObject::getId)
                .thenComparingInt(candidate -> candidate.getWorldLocation().getX())
                .thenComparingInt(candidate -> candidate.getWorldLocation().getY()));
            for (int index = startIndex; index < scanEnd; index++) {
                for (TileObject object : objects) {
                    if (object instanceof WallObject && Rs2DoorGeometry.wallDoorTouchesSegment(
                            (WallObject) object, route.get(index), route.get(index + 1))) {
                        return new Door(object, route.get(index));
                    }
                }
            }
            return null;
        }).orElse(null);
        WorldPoint approach = door != null ? door.before : transport != null ? transport.getOrigin() : walkTarget;
        if (door == null && transport != null && near(player, approach, 1) && !Rs2Player.isMoving()) {
            if (floor(transport.getDestination()) != floor(player) || !isShortcut(transport.getObjectId())) {
                recover("Planner selected an unsupported dungeon transport: " + transport.getDisplayInfo(), player);
                return false;
            }
            TileObject object = Rs2GameObject.findObjectByIdAndDistance(transport.getObjectId(), 4);
            if (object != null && Rs2GameObject.hasAction(object, transport.getAction()) && canInput()
                    && recordAttempt("shortcut:" + transport.getObjectId(), player)) {
                Rs2GameObject.interact(object, transport.getAction());
                interactionDelay();
                WorldPoint expected = transport.getDestination();
                resetPlan();
                pendingTransport = expected;
                pendingTransportAt = now();
            }
            return false;
        }
        if (door != null && near(player, approach, 1)
                && near(player, door.object.getWorldLocation(), 2)) {
            if (!Rs2Player.isMoving() && canInput()) {
                String action = Rs2GameObject.hasAction(door.object, "Open") ? "Open" : "Pass-through";
                if (!Rs2GameObject.hasAction(door.object, action)) {
                    recover("Unrecognized Stronghold door action.", player);
                    return false;
                }
                if (!recordAttempt("door:" + door.object.getId() + ":" + door.object.getWorldLocation(), player)) {
                    return false;
                }
                Rs2GameObject.interact(door.object, action);
                interactionDelay();
                resetPlan();
                pendingDoor = door;
                pendingDoorFrom = player;
                pendingDoorAt = now();
            }
        } else {
            if (stepTarget == null || !near(stepTarget, approach, 1)) {
                stepTarget = jitter(approach, player, true);
            }
            WalkerState result = Rs2Walker.walkStep(stepTarget, 0);
            if (result == WalkerState.UNREACHABLE || result == WalkerState.EXIT) {
                resetPlan();
            }
        }
        return false;
    }

    private void monitorSurface(WorldPoint player) {
        if (!player.equals(previousTile)) {
            previousTile = player;
            lastMovedAt = now();
        } else if (now() - lastMovedAt >= 8000) {
            Microbot.log("[Stronghold] No tile movement for 8 seconds; cancelling and reissuing walkTo.");
            surfaceTask.cancel(true);
            Rs2Walker.setTarget(null);
            lastMovedAt = now();
            // A subsequent cycle reissues only after the old worker's finally block ran.
        }
    }

    private WorldPoint resolveStand(WorldPoint base, WorldPoint player) {
        if (floor(player) < 0 || !near(base, player, 8) || Rs2Tile.isWalkable(base)) {
            return base;
        }
        List<WorldPoint> candidates = new ArrayList<>();
        for (int dx = -1; dx <= 1; dx++) {
            for (int dy = -1; dy <= 1; dy++) {
                WorldPoint candidate = new WorldPoint(base.getX() + dx, base.getY() + dy, base.getPlane());
                if (Rs2Tile.isWalkable(candidate)) {
                    candidates.add(candidate);
                }
            }
        }
        return candidates.stream().min(Comparator.comparingInt((WorldPoint candidate) ->
                Math.abs(candidate.getX() - base.getX()) + Math.abs(candidate.getY() - base.getY()))
            .thenComparingInt(WorldPoint::getX).thenComparingInt(WorldPoint::getY)).orElse(base);
    }

    private WorldPoint jitter(WorldPoint base, WorldPoint player, boolean requireReachable) {
        WorldPoint candidate = config.enableJitter()
            ? new WorldPoint(base.getX() + Rs2Random.between(-1, 2),
                base.getY() + Rs2Random.between(-1, 2), base.getPlane()) : base;
        if (floor(player) < 0) {
            return candidate;
        }
        boolean walkable = Rs2Tile.isWalkable(candidate);
        if (walkable && Rs2Tile.getReachableTilesFromTile(base, 4).containsKey(candidate)
                && (!requireReachable || Rs2Tile.getReachableTilesFromTile(player, 64).containsKey(candidate))) {
            return candidate;
        }
        return base;
    }

    private void handleDialogue(Snapshot snapshot) {
        if (state == State.HOMING_TO_BANK && !failure.isEmpty()) {
            if (canInput() && Rs2Dialogue.isInDialogue()) {
                Rs2Keyboard.keyPress(KeyEvent.VK_ESCAPE);
                interactionDelay();
            }
            return;
        }
        String text = normalize(Rs2Dialogue.getDialogueText());
        String question = normalize(Rs2Dialogue.getQuestion());
        if (!text.isEmpty()) {
            lastDialogue = text;
        }
        String answer = findAnswer(text + " " + question);
        if (answer != null) {
            rememberedAnswer = answer;
        }
        if (Rs2Dialogue.hasSelectAnOption()) {
            List<String> options = optionTexts();
            String choice = null;
            if (state == State.CLAIMING_REWARDS) {
                choice = options.stream().filter(option -> normalize(option).matches(".*\\bfancy boots\\b.*"))
                    .findFirst().orElse(null);
            }
            if (choice == null && state == State.CLAIMING_REWARDS && fancySelected
                    && (lastDialogue.contains("sure") || lastDialogue.contains("fancy boots"))) {
                choice = options.stream().filter(option -> normalize(option).startsWith("yes"))
                    .findFirst().orElse(null);
            }
            if (choice == null && rememberedAnswer != null) {
                final String expected = normalize(rememberedAnswer);
                choice = options.stream().filter(option -> normalize(option).equals(expected)).findFirst().orElse(null);
            }
            if (choice == null && descentPending && (lastDialogue.contains("danger")
                    || lastDialogue.contains("next level") || lastDialogue.contains("descend"))) {
                choice = options.stream().filter(option -> normalize(option).startsWith("yes"))
                    .findFirst().orElse(null);
            }
            if (choice == null) {
                finish(State.FAILED, "Unknown dialogue; no answer guessed. Question=" + question
                    + "; context=" + lastDialogue + "; options=" + options);
                return;
            }
            final String selected = choice;
            final List<String> before = options;
            readingDelay();
            if (canInput() && Rs2Dialogue.isInDialogue() && Rs2Dialogue.hasSelectAnOption()
                    && before.equals(optionTexts())
                    && question.equals(normalize(Rs2Dialogue.getQuestion()))) {
                boolean clicked = Rs2Dialogue.clickOption(selected, true);
                if (clicked && state == State.CLAIMING_REWARDS
                        && normalize(selected).matches(".*\\bfancy boots\\b.*")) {
                    fancySelected = true;
                }
                if (clicked) {
                    rememberedAnswer = null;
                }
                pendingDoorAt = now();
                interactionDelay();
            }
        } else if (Rs2Dialogue.hasContinue()) {
            readingDelay();
            if (canInput() && Rs2Dialogue.isInDialogue() && Rs2Dialogue.hasContinue()
                    && text.equals(normalize(Rs2Dialogue.getDialogueText()))) {
                Rs2Dialogue.clickContinue();
                pendingDoorAt = now();
                interactionDelay();
            }
        }
    }

    private List<String> optionTexts() {
        return Microbot.getClientThread().runOnClientThreadOptional(() -> {
            List<String> result = new ArrayList<>();
            for (Widget option : Rs2Dialogue.getDialogueOptions()) {
                result.add(option.getText());
            }
            return result;
        }).orElse(Collections.emptyList());
    }

    private static String findAnswer(String question) {
        for (Map.Entry<String, String> entry : SECURITY_ANSWERS.entrySet()) {
            if (question.contains(entry.getKey())) {
                return entry.getValue();
            }
        }
        return null;
    }

    private boolean expectedZone(WorldPoint tile) {
        switch (state) {
            case VALIDATING_ANCHOR:
                return near(tile, ANCHOR, 15);
            case WALKING_TO_STRONGHOLD:
                return inRectangle(tile, 3060, 3400, 3190, 3510, 15);
            case INTERACTING_ENTRANCE:
                return near(tile, ENTRANCE, 15) || inFloor(tile, 0, 15);
            case TRAVERSING_FLOOR_1:
            case TRAVERSING_FLOOR_2:
            case TRAVERSING_FLOOR_3:
            case TRAVERSING_FLOOR_4:
                int index = state.ordinal() - State.TRAVERSING_FLOOR_1.ordinal();
                return inFloor(tile, index, 15)
                    || (descentPending && index < 3 && inFloor(tile, index + 1, 15));
            case CLAIMING_REWARDS:
                return inFloor(tile, 3, 15);
            case HOMING_TO_BANK:
                for (int floorIndex = 0; floorIndex < 4; floorIndex++) {
                    if (inFloor(tile, floorIndex, 15)) {
                        return true;
                    }
                }
                int x = recoveryOrigin != null && floor(recoveryOrigin) < 0 ? recoveryOrigin.getX() : 3081;
                int y = recoveryOrigin != null && floor(recoveryOrigin) < 0 ? recoveryOrigin.getY() : 3421;
                return inRectangle(tile, Math.min(x, 3060) - 32, Math.min(y, 3400) - 32,
                    Math.max(x, 3190) + 32, Math.max(y, 3510) + 32, 15);
            case DEPOSITING_LOOT:
                return near(tile, ANCHOR, 15);
            case COMPLETED:
            case FAILED:
                return true;
            default:
                return false;
        }
    }

    private void recover(String reason, WorldPoint location) {
        if (failure.isEmpty()) {
            failure = reason;
        }
        rewardsVerified = false;
        recoveryOrigin = location;
        transition(State.HOMING_TO_BANK, "Recovery: " + reason);
    }

    private void transition(State next, String reason) {
        State previous = state;
        cancelNavigation();
        state = next;
        enteredAt = now();
        actionKey = "";
        actionAttempts = 0;
        descentPending = false;
        rememberedAnswer = null;
        lastDialogue = "";
        Microbot.log("[Stronghold] " + previous + " -> " + next + " | " + reason);
        journal(reason);
    }

    private void finish(State terminal, String reason) {
        if (!active) {
            return;
        }
        if (terminal == State.FAILED && failure.isEmpty()) {
            failure = reason;
        }
        transition(terminal, reason);
        shutdown();
    }

    private void journal(String reason) {
        if (!account.isEmpty()) {
            String key = "checkpoint_" + account.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]", "_");
            try {
                configManager.setConfiguration(StrongholdConfig.GROUP, key,
                    Instant.now() + " | account=" + account + " | state=" + state
                        + " | verified=" + rewardsVerified + " | baselineCoins=" + baselineCoins
                        + " | reason=" + reason.replace('\n', ' '));
            } catch (RuntimeException exception) {
                Microbot.log("[Stronghold] Checkpoint persistence failed: " + exception);
            }
        }
    }

    private void resetPlan() {
        Rs2Walker.setTarget(null);
        plannedPath = Collections.emptyList();
        plannedAt = 0;
        stepTarget = null;
        pendingDoor = null;
        pendingTransport = null;
        plannedRoute = null;
    }

    private void cancelNavigation() {
        if (surfaceTask != null && !surfaceTask.isDone()) {
            surfaceTask.cancel(true);
        }
        resetPlan();
        destination = null;
        walkTarget = null;
    }

    @Override
    public synchronized void shutdown() {
        if (active && state != State.COMPLETED && state != State.FAILED) {
            State previous = state;
            state = State.FAILED;
            failure = "Plugin stopped before completion.";
            Microbot.log("[Stronghold] " + previous + " -> FAILED | " + failure);
            journal(failure);
        }
        active = false;
        cancelNavigation();
        if (walkingExecutor != null) {
            walkingExecutor.shutdownNow();
        }
        super.shutdown();
        if (ownsWalkerSettings) {
            Rs2Walker.disableTeleports = oldDisableTeleports;
            ownsWalkerSettings = false;
        }
    }

    private boolean canInput() {
        return active && !Thread.currentThread().isInterrupted() && Microbot.isLoggedIn()
            && !Microbot.pauseAllScripts.get() && !InputArbiter.isHuman();
    }

    private long stateBudget() {
        switch (state) {
            case TRAVERSING_FLOOR_1:
            case TRAVERSING_FLOOR_2:
            case TRAVERSING_FLOOR_3:
            case TRAVERSING_FLOOR_4:
            case HOMING_TO_BANK:
                return TimeUnit.MINUTES.toMillis(10);
            default:
                return TimeUnit.MINUTES.toMillis(2);
        }
    }

    private Snapshot snapshot() {
        return Microbot.getClientThread().runOnClientThreadOptional(() -> {
            if (Microbot.getClient().getLocalPlayer() == null) {
                return null;
            }
            Snapshot result = new Snapshot();
            result.tile = Microbot.getClient().getLocalPlayer().getWorldLocation();
            result.name = Microbot.getClient().getLocalPlayer().getName();
            if (result.name == null) {
                return null;
            }
            result.hp = Microbot.getClient().getBoostedSkillLevel(Skill.HITPOINTS);
            result.maxHp = Microbot.getClient().getRealSkillLevel(Skill.HITPOINTS);
            for (int index = 0; index < REWARD_BITS.length; index++) {
                result.claimed[index] = Microbot.getClient().getVarbitValue(REWARD_BITS[index]) != 0;
            }
            ItemContainer inventory = Microbot.getClient().getItemContainer(InventoryID.INVENTORY);
            ItemContainer equipment = Microbot.getClient().getItemContainer(InventoryID.EQUIPMENT);
            if (inventory == null || equipment == null) {
                return null;
            }
            result.inventoryEmpty = empty(inventory);
            result.equipmentEmpty = empty(equipment);
            for (Item item : inventory.getItems()) {
                if (item.getId() == COINS) {
                    result.coins += item.getQuantity();
                } else if (item.getId() == FANCY_BOOTS) {
                    result.boots += item.getQuantity();
                }
            }
            return result;
        }).orElse(null);
    }

    private static boolean empty(ItemContainer container) {
        for (Item item : container.getItems()) {
            if (item.getId() >= 0 && item.getQuantity() > 0) {
                return false;
            }
        }
        return true;
    }

    private static boolean isShortcut(int id) {
        return id == ObjectID.SOS_WAR_PORTAL || id == ObjectID.SOS_FAM_PORTAL
            || id == ObjectID.SOS_PEST_PORTAL || id == ObjectID.SOS_DEATH_PORTAL
            || id == ObjectID.SOS_FAM_ROPE_UP || id == ObjectID.SOS_PEST_ROPE_UP || id == 1734;
    }

    private static boolean isStrongholdDoor(int id) {
        return id == ObjectID.SOS_WAR_DOOR_FACE || id == ObjectID.SOS_WAR_DOOR_FACE_MIRR
            || id == ObjectID.SOS_FAM_DOOR_FACE || id == ObjectID.SOS_FAM_DOOR_FACE_MIRR
            || id == ObjectID.SOS_PEST_DOOR_FACE || id == ObjectID.SOS_PEST_DOOR_FACE_MIRR
            || id == ObjectID.SOS_DEATH_DOOR_FACE || id == ObjectID.SOS_DEATH_DOOR_FACE_MIRR;
    }

    private static WorldPoint point(int x, int y) {
        return new WorldPoint(x, y, 0);
    }

    private static boolean near(WorldPoint first, WorldPoint second, int radius) {
        return first != null && second != null && first.getPlane() == second.getPlane()
            && first.distanceTo2D(second) <= radius;
    }

    private static boolean inRectangle(WorldPoint point, int minX, int minY, int maxX, int maxY, int margin) {
        return point != null && point.getPlane() == 0 && point.getX() >= minX - margin
            && point.getX() <= maxX + margin && point.getY() >= minY - margin && point.getY() <= maxY + margin;
    }

    private static boolean inFloor(WorldPoint point, int index, int margin) {
        int[] bounds = FLOOR_BOUNDS[index];
        return inRectangle(point, bounds[0], bounds[1], bounds[2], bounds[3], margin);
    }

    private static int floor(WorldPoint point) {
        for (int index = 0; index < FLOOR_BOUNDS.length; index++) {
            if (inFloor(point, index, 0)) {
                return index;
            }
        }
        return -1;
    }

    private static long now() {
        return TimeUnit.NANOSECONDS.toMillis(System.nanoTime());
    }

    private static String normalize(String value) {
        return value == null ? "" : value.replaceAll("(?i)<br\\s*/?>", " ")
            .replaceAll("<[^>]*>", "").replace('\u00a0', ' ').replace('\u2019', '\'')
            .replaceAll("\\s+", " ").trim().toLowerCase(Locale.ROOT);
    }

    private static void interactionDelay() {
        // Global.sleep is inherited by Script; Microbot.sleep does not exist in this revision.
        sleep(Rs2Random.between(1200, 2800));
    }

    private static void readingDelay() {
        sleep(Rs2Random.between(1800, 4200));
    }

package net.runelite.client.plugins.microbot.tradeunlock;

import java.awt.Color;
import java.awt.Dimension;
import java.awt.Graphics2D;
import java.util.concurrent.TimeUnit;
import javax.inject.Inject;
import javax.inject.Singleton;
import net.runelite.client.ui.overlay.OverlayPanel;
import net.runelite.client.ui.overlay.OverlayPosition;
import net.runelite.client.ui.overlay.components.LineComponent;
import net.runelite.client.ui.overlay.components.TitleComponent;

@Singleton
public final class TradeUnlockOverlay extends OverlayPanel {
    private final TradeUnlockScript script;
    private final TradeUnlockConfig config;

    @Inject
    public TradeUnlockOverlay(TradeUnlockScript script, TradeUnlockConfig config) {
        this.script = script;
        this.config = config;
        setPosition(OverlayPosition.TOP_LEFT);
        panelComponent.setPreferredSize(new Dimension(300, 0));
    }

    @Override
    public Dimension render(Graphics2D graphics) {
        if (!config.showOverlay()) { return null; }
        TradeUnlockScript.Status status = script.getStatus();
        panelComponent.getChildren().clear();
        panelComponent.getChildren().add(TitleComponent.builder()
            .text("Stage 0.2 | F2P Quests").color(new Color(120, 210, 230)).build());
        line("State", status.state.replace('_', ' '));
        line("Quest", status.quest);
        line("Objective", status.objective);
        line("Quest points", Integer.toString(status.questPoints));
        line("QP gained this run", Integer.toString(status.earnedPoints));
        long seconds = TimeUnit.MILLISECONDS.toSeconds(status.runtimeMillis);
        line("Runtime", String.format("%02d:%02d:%02d", seconds / 3600, (seconds / 60) % 60, seconds % 60));
        panelComponent.getChildren().add(LineComponent.builder().left("Status")
            .right(status.detail).rightColor("FAILED".equals(status.state) ? Color.RED : Color.WHITE).build());
        return super.render(graphics);
    }

    private void line(String left, String right) {
        panelComponent.getChildren().add(LineComponent.builder().left(left).right(right).build());
    }
}
